# Macros do plugin "macro" (ja carregado no sys.txt).

# Rebuff: quando Bencao e Agilidade acabam no mapa de farm, volta para Prontera com @go 0.
# O servidor buffa automaticamente ao chegar na cidade e o lockMap traz o personagem de
# volta pela Teletransportadora (prontera 163 190, ver portals_ragnawicked.txt).
# timeout evita repetir o comando caso o @go demore ou falhe.
automacro rebuffEmProntera {
	map anthell02
	status not Blessing
	status not Increase Agility
	timeout 30
	call irParaProntera
}

macro irParaProntera {
	log [rebuff] Buffs acabaram, voltando para Prontera com @go 0
	do c @go 0
}

# O Matador nao senta para descansar (sitAuto_* 0 no config.txt): com pouca vida usa
# @go 0, o servidor cura ao chegar em Prontera e o lockMap traz de volta.
automacro vidaBaixa {
	map anthell02
	hp < 40%
	timeout 20
	call curarEmProntera
}

# Depois de morrer ele renasce em Prontera com pouca vida: @go 0 para receber a cura.
automacro renascidoSemVida {
	map prontera
	hp < 80%
	timeout 30
	call curarEmProntera
}

macro curarEmProntera {
	log [recuperar] HP baixo, @go 0 para o servidor curar
	do c @go 0
}
