# Macros do plugin "macro" (ja carregado no sys.txt).

# @autoloot a cada entrada no jogo (inclusive apos reconectar).
# "@autoloot 100" define a taxa (nao alterna), entao repetir e seguro.
automacro autolootAoLogar {
	console /You are now in the game/
	timeout 5
	call ligarAutoloot
}

macro ligarAutoloot {
	pause 3
	do c @autoloot 100
}

# Pouca vida (a autocura nao deu conta): volta para Prontera com @go 0.
# O servidor cura e buffa ao chegar na cidade; o lockMap traz de volta pela
# Teletransportadora (prontera 163 190, ver portals_ragnawicked.txt).
automacro fugirVidaBaixa {
	map gef_fild14
	hp < 30%
	timeout 20
	call irParaProntera
}

# Sem SP nao ha Cura (nem para atacar, nem para se curar): mesma saida.
automacro fugirSemSP {
	map gef_fild14
	sp < 10%
	timeout 20
	call irParaProntera
}

macro irParaProntera {
	log [recuperar] HP/SP baixo, voltando para Prontera com @go 0
	do c @go 0
}

# Follow x lockMap: com o follow ligado a IA nunca fica ociosa e o lockMap nao age.
# Em Prontera (depois de morrer ou do @go 0) desliga o follow para o lockMap levar
# de volta pela Teletransportadora; no mapa do grupo, liga de novo.
automacro voltarSemSeguir {
	map prontera
	timeout 5
	call naoSeguir
}

macro naoSeguir {
	if (@config(follow) == 0) goto fim
	log [grupo] Em Prontera: follow desligado para voltar a gef_fild14
	do conf follow 0
	:fim
}

automacro seguirNoMapa {
	map gef_fild14
	timeout 5
	call seguir
}

macro seguir {
	if (@config(follow) == 1) goto fim
	log [grupo] De volta a gef_fild14: seguindo o King Slayer
	do conf follow 1
	:fim
}
