# Macros do plugin "macro" (ja carregado no sys.txt).
# Suporte do King Slayer: os buffs vem da Ashiba (party), entao so a vida baixa tira
# o Obss do mapa. @go 0 leva a Prontera (o servidor cura ao chegar) e o lockMap traz
# de volta pela Teletransportadora (prontera 163 190, ver portals_ragnawicked.txt).
automacro vidaBaixa {
	map alde_dun02
	hp < 40%
	timeout 20
	call irParaProntera
}

macro irParaProntera {
	log [recuperar] HP baixo, voltando para Prontera com @go 0
	do c @go 0
}

# Follow x lockMap: com o follow ligado a IA nunca fica ociosa e o lockMap nao age.
# Em Prontera (depois de morrer ou do @go 0) desliga o follow para o lockMap levar
# de volta pela Teletransportadora; no mapa do grupo, liga de novo.
automacro voltarSemSeguir {
	map prontera
	timeout 5
	call naoSeguir
}

macro naoSeguir {
	if (@config(follow) == 0) goto fim
	log [grupo] Em Prontera: follow desligado para voltar a alde_dun02
	do conf follow 0
	:fim
}

automacro seguirNoMapa {
	map alde_dun02
	timeout 5
	call seguir
}

macro seguir {
	if (@config(follow) == 1) goto fim
	log [grupo] De volta a alde_dun02: seguindo o King Slayer
	do conf follow 1
	:fim
}
