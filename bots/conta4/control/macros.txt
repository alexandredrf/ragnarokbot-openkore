# Macros do plugin "macro" (ja carregado no sys.txt).
# Suporte do King Slayer: os buffs vem da Ashiba (party), entao so a vida baixa tira
# o Obss do mapa. @go 0 leva a Prontera (o servidor cura ao chegar) e o lockMap traz
# de volta pela Teletransportadora (prontera 163 190, ver portals_ragnawicked.txt).
automacro vidaBaixa {
	map gef_fild14
	hp < 40%
	timeout 20
	call irParaProntera
}

macro irParaProntera {
	log [recuperar] HP baixo, voltando para Prontera com @go 0
	do c @go 0
}
